# IDFC GenAI - TRAIN/TEST EXPERIMENTS
# Two splits: 395/100 and 412/83
# Final system: Accepts PDF or Image input
# ============================================================

# ============================================================
# SPLIT 1: 395 train, 100 test (80/20)
# ============================================================

# Step 1.1: Create split
python3 split_dataset.py --train-size 395

# Step 1.2: Build YOLO dataset from split
python3 build_yolo_from_split.py --split-dir splits/split_395_100

# Step 1.3: Train YOLO on split 1
yolo detect train data=splits/split_395_100/yolo_data/data.yaml model=yolo11m.pt epochs=200 imgsz=1024 batch=2 device=0 project=splits/split_395_100 name=yolo patience=30

# Step 1.4: Build VLM training data from split
python3 build_vlm_from_split.py --split-dir splits/split_395_100

# Step 1.5: Train VLM on split 1
python3 vlm_lora_train_v2.py --model-path models/qwen2.5-vl-7b-instruct --train-file splits/split_395_100/vlm_train.jsonl --output-dir splits/split_395_100/vlm_lora --batch-size 1 --accum 8 --epochs 10 --lr 5e-5 --lora-r 64 --lora-alpha 128 --device 0 --load-4bit

# Step 1.6: Evaluate on test set (100 images)
python3 evaluate.py --split-dir splits/split_395_100 --yolo-model splits/split_395_100/yolo/weights/best.pt --use-lora --device 0


# ============================================================
# SPLIT 2: 412 train, 83 test (~83/17)
# ============================================================

# Step 2.1: Create split
python3 split_dataset.py --train-size 412

# Step 2.2: Build YOLO dataset from split
python3 build_yolo_from_split.py --split-dir splits/split_412_83

# Step 2.3: Train YOLO on split 2
yolo detect train data=splits/split_412_83/yolo_data/data.yaml model=yolo11m.pt epochs=200 imgsz=1024 batch=2 device=0 project=splits/split_412_83 name=yolo patience=30

# Step 2.4: Build VLM training data from split
python3 build_vlm_from_split.py --split-dir splits/split_412_83

# Step 2.5: Train VLM on split 2
python3 vlm_lora_train_v2.py --model-path models/qwen2.5-vl-7b-instruct --train-file splits/split_412_83/vlm_train.jsonl --output-dir splits/split_412_83/vlm_lora --batch-size 1 --accum 8 --epochs 10 --lr 5e-5 --lora-r 64 --lora-alpha 128 --device 0 --load-4bit

# Step 2.6: Evaluate on test set (83 images)
python3 evaluate.py --split-dir splits/split_412_83 --yolo-model splits/split_412_83/yolo/weights/best.pt --use-lora --device 0


# ============================================================
# AFTER EXPERIMENTS: Train final model on ALL 495 images
# ============================================================

# Final YOLO (all 495)
python3 build_yolo_dataset.py
yolo detect train data=yolo_data/data.yaml model=yolo11m.pt epochs=200 imgsz=1024 batch=2 device=0 project=runs/detect name=stamp_sig_final patience=30

# Final VLM (all 495)
python3 prepare_vlm_jsonl.py
python3 vlm_lora_train_v2.py --model-path models/qwen2.5-vl-7b-instruct --train-file data/vlm_gold.jsonl --output-dir vlm_lora_out --batch-size 1 --accum 8 --epochs 10 --lr 5e-5 --lora-r 64 --lora-alpha 128 --device 0 --load-4bit
